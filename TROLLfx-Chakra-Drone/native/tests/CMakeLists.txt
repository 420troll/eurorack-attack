cmake_minimum_required(VERSION 3.16)
project(ChakraTuningTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(chakra_pitch_test test_pitch.cpp)
target_include_directories(chakra_pitch_test PRIVATE ../include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(chakra_pitch_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME chakra_tuning COMMAND chakra_pitch_test)
