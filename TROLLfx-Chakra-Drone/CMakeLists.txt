cmake_minimum_required(VERSION 3.16)
project(TROLLfxChakraDrone VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(chakra_render src/chakra_render.cpp)
if(MSVC)
  target_compile_options(chakra_render PRIVATE /W4)
else()
  target_compile_options(chakra_render PRIVATE -Wall -Wextra -Wpedantic)
endif()
